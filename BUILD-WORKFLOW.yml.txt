# Igual ao workflow do JPG Diet (verde): SDK 36 e Gradle 8.11.1.
# Sem NDK/CMake: o DX PTT nao tem codigo nativo.
# ============================================================

name: Build APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest

    env:
      ANDROID_HOME: /usr/local/lib/android/sdk
      ANDROID_SDK_ROOT: /usr/local/lib/android/sdk

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Instalar Android SDK 36
        run: |
          SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          yes | "$SDKMANAGER" --licenses > /dev/null || true
          "$SDKMANAGER" "platforms;android-36" "build-tools;36.0.0"

      - name: Setup Gradle 8.11.1
        uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: '8.11.1'

      - name: Build debug APKs
        run: gradle assembleDebug --console=plain

      - name: Upload do APK (celular)
        uses: actions/upload-artifact@v4
        with:
          name: dx-ptt-phone-apk
          path: app/build/outputs/apk/debug/app-debug.apk

      - name: Upload do APK (relogio)
        uses: actions/upload-artifact@v4
        with:
          name: dx-ptt-watch-apk
          path: wear/build/outputs/apk/debug/wear-debug.apk
